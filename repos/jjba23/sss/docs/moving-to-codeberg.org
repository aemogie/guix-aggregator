#+title: Moving to Codeberg

As I stand for my freedom and that of others, and want to avoid any ties to evil big corporations that promote proprietary software, I moved to Codeberg.

#+begin_src sh
  function ow_github_with_move_to_codeberg() {
      cd ~/Ontwikkeling/Scratch/
      git clone "git@personal.github.com:jjba23/$1.git"
      cd "$1"
      rm -rf ./*
      rm -rf ./.*
      git init
      git branch -m trunk
      echo "* Moved to Codeberg: https://codeberg.org/jjba23/$2" >> README.org
      git add -A
      git commit -am "migrated"
      git remote add github "git@personal.github.com:jjba23/$1.git"
      git push -uf github trunk
  }
#+end_src

Use it with:
#+begin_src sh 
  ow_github_with_move_to_codeberg honeybadger honeybadger
#+end_src
