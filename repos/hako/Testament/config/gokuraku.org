  (simple-service 'setup-mirror
      shepherd-root-service-type
    (list (shepherd-timer '(setup-mirror)
            #~(calendar-event #:hours '(0 6 12 18) #:minutes '(0))
            #~(#$(program-file "setup-mirror"
                   (with-imported-modules '((guix build utils))
                     #~(begin
                         (use-modules (guix build utils))
                         ;; Upload storage to Cloudflare R2.
                         (invoke "/bin/sh"
                                 #$(local-file (testament-plain "gokuraku/sync-r2.sh"))
                                 #$(file-append (spec->pkg "rclone-bin") "/bin/rclone"))
                         ;; Set up narinfo mirror.
                         (mkdir-p "/var/cache/guix/narinfo")
                         (invoke #$(file-append (spec->pkg "rsync") "/bin/rsync")
                                 "/var/cache/guix/publish/" "/var/cache/guix/narinfo/"
                                 "-avh"
                                 "--delete"
                                 "--exclude" "/hashes"
                                 "--exclude" "/nar")))))
            #:requirement '(user-processes))))
