#+begin_src scheme :tangle mirror.scm :noweb yes
  (use-modules <<module>>)

  (define (%os mirror-name bios-boot)
    (operating-system
      (host-name (string-append mirror-name ".guix.moe"))
      (timezone "Etc/GMT-8")
      (bootloader
       (bootloader-configuration
         (bootloader
           (if bios-boot
               grub-bootloader
               grub-efi-removable-bootloader))
         (targets (list (or bios-boot "/efi")))))
      (kernel linux-lts/dolly)
      (kernel-arguments
       (cons* "kernel.sysrq=0"
              "zswap.enabled=1"
              "zswap.max_pool_percent=90"
              %default-kernel-arguments))
      (initrd-modules
       (cons* "sd_mod"
              "virtio_scsi"
              %base-initrd-modules))
      (file-systems (append <<file-systems>> %base-file-systems))
      (swap-devices
       (list (swap-space
               (target (file-system-label "swap")))))
      (users (cons* <<user>> %base-user-accounts))
      (packages (append (specs->pkgs+out "mosh" "rsync") %base-packages))
      (services
       (cons* <<service>>

              (simple-service 'extend-guix guix-service-type
                (guix-extension
                  (authorized-keys %guix-keys)
                  (substitute-urls '("https://cache-cdn.guix.moe"))))

              (simple-service 'guix-gc shepherd-root-service-type
                (list (shepherd-timer '(guix-gc)
                        #~(calendar-event #:days-of-week '(sunday) #:hours '(12) #:minutes '(0))
                        #~("/run/current-system/profile/bin/guix" "gc" "--delete-generations=1m")
                        #:requirement '(user-processes guix-daemon))))

              %base-services))
      (sudoers-file
       (plain-file "sudoers"
         (string-append
          (plain-file-content %sudoers-specification)
          "%wheel ALL = NOPASSWD: ALL\n")))))

  %os
#+end_src

*** mirroring
#+begin_src scheme :noweb-ref service
  (simple-service 'guix-moe-mirror shepherd-root-service-type
    (list (shepherd-timer '(guix-moe-mirror)
            #~(calendar-event #:minutes '(45))
            #~(#$(program-file "guix-moe-mirror"
                   (with-imported-modules
                       (source-module-closure
                        '((guix build syscalls)
                          (guix build utils)))
                     #~(begin
                         (use-modules (guix build syscalls)
                                      (guix build utils))
                         (with-file-lock/no-wait "/var/lock/guix-moe-mirror.lock"
                           (lambda _
                             (error "found ongoing syncing process, exiting"))
                           (begin
                             (mkdir-p "/var/cache/guix-moe/narinfo")
                             (invoke #$(file-append (spec->pkg "rsync") "/bin/rsync")
                                     "-az" "--delete"
                                     "rsync://cache.guix.moe/narinfo/"
                                     "/var/cache/guix-moe/narinfo/")))))))
            #:requirement '(user-processes))))
#+end_src
