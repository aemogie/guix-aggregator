  (load "../common.scm")
  (use-modules (rosenthal) <<module>>)

  (define %os
    (operating-system
      (host-name "nuporta")
      (timezone "Etc/GMT-8")
      ;; GRUB has issue booting from the ZFS dataset with /gnu/store.
      ;; See https://savannah.gnu.org/bugs/?47742 and issues with same error.
      ;; NOTE: This bootloader currently doesn't support choosing system
      ;; generation on boot and switching system generations / rolling back.
      (bootloader
       (bootloader-configuration
         (bootloader uefi-uki-removable-bootloader)
         (targets '("/efi"))))
      (kernel linux-lts/dolly)
      (kernel-arguments
       (cons* "kernel.sysrq=0"
              "zswap.enabled=1"
              "zswap.max_pool_percent=90"
              %default-kernel-arguments))
      (file-systems (append <<file-systems>> %base-file-systems))
      (swap-devices
       (list (swap-space
               (target (file-system-label "swap")))))
      (users (cons* <<user>> %base-user-accounts))
      (packages
       (append (specs->pkgs+out
                "curl"
                "fd"
                "gnupg"
                "mosh"
                "rclone"
                "ripgrep"
                "rsync"
                "sops"
                <<package>>)
               %base-packages))
      (services
       (cons* <<service>>

              (service guix-publish-service-type
                (guix-publish-configuration
                  (guix (spec->pkg "guix-dolly"))
                  (port 49637)
                  (compression '(("zstd" 19)))
                  (cache "/var/cache/guix/publish")
                  ;; 180 days
                  (ttl (* 180 24 3600))
                  ;; 2 minutes
                  (negative-ttl (* 2 60))))

              (simple-service 'extend-guix guix-service-type
                (guix-extension
                  (authorized-keys %testament-guix-keys)
                  (substitute-urls
                   '("https://cache-fi.guix.moe"
                     "https://substitutes.nonguix.org"))))

              (modify-services %base-services
                (guix-service-type
                 config => (guix-configuration
                             (inherit config)
                             (extra-options '("--cores=4")))))))
      (privileged-programs
       (cons* (privileged-program
                (program (file-append manage-cuirass "/bin/manage-cuirass"))
                (setuid? #t))
              %default-privileged-programs))))

  ((compose (rosenthal-transformation-zfs #:boot? #t))
   %os)
