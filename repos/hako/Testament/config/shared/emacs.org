** Interface
*** [[https://github.com/Wilfred/helpful][helpful]]
Alternative to the built-in Emacs help that provides much more contextual
information.

#+begin_src scheme :noweb-ref package-spec
  "emacs-helpful"
  "emacs-pgtk:doc"
#+end_src

Source code of Emacs is optional.  It will be used for displaying source of C
functions.

In Guix, =doc= output of the Emacs package stores source code under the
=c-source= directory, differing from =src= expected by =helpful--library-path=.
Redefine the function as a workaround.

#+begin_src emacs-lisp :noweb-ref emacs-init
  (use-package helpful
    :config
    (defun helpful--library-path (library-name)
      (when (member (f-ext library-name) '("c" "rs"))
        (setq library-name
              (f-expand (f-filename library-name)
                        find-function-C-source-directory)))
      (condition-case nil
          (find-library-name library-name)
        (error nil)))
    :bind
    ([remap describe-function] . helpful-callable)
    ([remap describe-variable] . helpful-variable)
    ([remap describe-key]      . helpful-key)
    ([remap describe-command]  . helpful-command)
    ([remap describe-symbol]   . helpful-symbol)
    ("C-c C-d" . helpful-at-point))
#+end_src

Xref stops searching if one backend fails to find the symbol.  Replace it with
Citre since the latter supports [[https://github.com/universal-ctags/citre/blob/master/docs/user-manual/adapt-an-existing-xref-backend.md][adapting existing Xref backends]] and handles
multiple backends better.

#+begin_src emacs-lisp :noweb-ref emacs-init
  (use-package citre
    :custom
    (citre-default-create-tags-file-location 'global-cache)
    (citre-edit-ctags-options-manually nil)
    (citre-find-definition-backends
     '(eglot elisp arei tags dumb-jump))
    (citre-find-reference-backends
     '(eglot elisp dumb-jump))
    :bind
    ([remap xref-find-definitions] . citre-jump)
    ([remap xref-find-references]  . citre-jump-to-reference)
    ([remap xref-go-back]          . citre-jump-back)
    ("C-x c p" . citre-ace-peek)
    ("C-x c u" . citre-update-this-tags-file))

  (use-package citre-config)
#+end_src

Citre has builtin support for Eglot.  Other backends will be converted later
in their own sections.

Elisp Xref backend doesn't handle =*.el.gz= files by itself and returns =*.el=
files instead.  Redefine so that Citre can find them.

#+begin_src emacs-lisp :noweb-ref emacs-init
  (use-package elisp-mode
    :after citre
    :config
    (defun elisp--xref-make-xref (type symbol file &optional summary)
      (let ((file* (if (string-suffix-p ".el" file)
                       (find-library-name file)
                     file)))
        (xref-make (or summary
  		     (format elisp--xref-format (or type 'defun) symbol))
  	         (xref-make-elisp-location symbol type file*))))
    (defvar citre-elisp-backend
      (citre-xref-backend-to-citre-backend
       'elisp
       (lambda () (derived-mode-p 'emacs-lisp-mode))))
    (citre-register-backend 'elisp citre-elisp-backend))
#+end_src

Arei Xref backend provides =xref-buffer-location= but Citre expects
=xref-file-location=.  Redefine it:

#+begin_src emacs-lisp :noweb-ref emacs-init
  (use-package arei-xref
    :after citre
    :config
    (defun arei-xref--make-xref-loc (identifier)
      (when-let* ((info (arei-xref--request-lookup identifier))
                  (file (arei-nrepl-dict-get info "file"))
                  (line (arei-nrepl-dict-get info "line"))
                  (column (arei-nrepl-dict-get info "column")))
        (xref-make-file-location file line column)))
    (defvar citre-arei-backend
      (citre-xref-backend-to-citre-backend
       'arei
       #'arei-connected-p))
    (citre-register-backend 'arei citre-arei-backend))
#+end_src

#+begin_src scheme :noweb-ref package-spec
  "fontconfig"
  "font-adobe-source-serif"
  "font-chiron-sung-hk"
  "font-google-noto-emoji"
#+end_src

#+begin_src emacs-lisp :noweb-ref emacs-fonts
  (defvar nema--font-size 12 "Font size")
  (defvar nema-fonts '((sans     . "sans")
                       (serif    . "serif")
                       (mono     . "monospace")
                       (cjk      . "Chiron Sung HK VF")
                       (symbol   . "Noto Color Emoji")
                       (modeline . "Source Serif 4"))
    "Fonts to use.")

  (defun nema//get-font-family (key)
    (alist-get key nema-fonts))

  (defun nema//generate-font-spec (key)
    (format "%s-%d"
            (nema//get-font-family key)
            nema--font-size))

  (defun nema//load-base-font ()
    "Load the default font for ascii characters."
    (let* ((font-spec (nema//generate-font-spec 'mono)))
      (set-frame-parameter nil 'font font-spec)
      (add-to-list 'default-frame-alist (cons 'font font-spec))))

  (defun nema//load-face-font ()
    "Load fonts used in faces.

  This function must be called after frame creation."
    (let ((mono (nema//generate-font-spec 'mono))
          (sans (nema//generate-font-spec 'sans))
          (serif (nema//generate-font-spec 'sans-serif))
          (modeline (nema//generate-font-spec 'modeline)))
      (set-face-attribute 'variable-pitch nil :font sans)
      (set-face-attribute 'variable-pitch-text nil :font serif)
      (set-face-attribute 'fixed-pitch nil :font mono)
      (set-face-attribute 'fixed-pitch-serif nil :font mono)
      (set-face-attribute 'mode-line nil :font modeline)
      (set-face-attribute 'mode-line-inactive nil :font modeline)))

  (defun nema//load-ext-font ()
    "Load fonts used for non-ascii characters.

  This function must be called after frame creation."
    (let ((font (frame-parameter nil 'font))
          (font-spec-cjk (font-spec :family (nema//get-font-family 'cjk)))
          (font-spec-symbol (font-spec :family (nema//get-font-family 'symbol))))
      (dolist (charset '(kana han hangul cjk-misc bopomofo))
        (set-fontset-font font charset font-spec-cjk))
      (set-fontset-font font 'symbol font-spec-symbol)))

  (defun nema/load-font ()
    "Load all font configuration."
    (interactive)
    (when (display-graphic-p)
      (nema//load-base-font)
      (nema//load-ext-font)
      (nema//load-face-font)))

  (if (daemonp)
      (add-hook 'server-after-make-frame-hook #'nema/load-font)
    ;; Else: not in daemon
    (add-hook 'after-init-hook #'nema/load-font))
#+end_src
