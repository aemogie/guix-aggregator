#+begin_src scheme :tangle chapra.scm :noweb yes
  (use-modules <<module>>)

  (define %os
    (operating-system
      (host-name "chapra")
      (timezone "Etc/GMT-8")
      ;; GRUB has issue booting from the ZFS dataset with /gnu/store.
      ;; See https://savannah.gnu.org/bugs/?47742 and issues with same error.
      ;; NOTE: This bootloader currently doesn't support choosing system
      ;; generation on boot and switching system generations / rolling back.
      (bootloader
       (bootloader-configuration
         (bootloader uefi-uki-removable-bootloader)
         (targets '("/efi"))))
      (kernel linux-lts/dolly)
      (kernel-arguments
       (cons* "kernel.sysrq=0"
              "memtest=3"
              "zswap.enabled=1"
              "zswap.max_pool_percent=90"
              %default-kernel-arguments))
      (file-systems (append <<file-systems>> %base-file-systems))
      (swap-devices
       (list (swap-space
               (target
                (uuid "b955a6b1-9dd9-4e91-b9d9-b4e15473396c")))))
      (users (cons* <<user>> %base-user-accounts))
      (packages
       (append (specs->pkgs+out
                "curl"
                "fd"
                "gnupg"
                "mosh"
                "ripgrep"
                "rsync"
                "sops"
                <<package>>)
               %base-packages))
      (services
       (cons* <<service>>

              (simple-service 'extend-guix guix-service-type
                (guix-extension
                  (authorized-keys %guix-keys)
                  (substitute-urls '("https://cache-cdn.guix.moe"))))

              (simple-service 'guix-gc shepherd-root-service-type
                (list (shepherd-timer '(guix-gc)
                        #~(calendar-event #:days-of-week '(sunday) #:hours '(12) #:minutes '(0))
                        #~("/run/current-system/profile/bin/guix" "gc" "--delete-generations=1m")
                        #:requirement '(user-processes guix-daemon))))

              (modify-services %base-services
                (guix-service-type
                 config => (guix-configuration
                             (inherit config)
                             (discover? #t))))))
      (name-service-switch %mdns-host-lookup-nss)))

  ((compose (rosenthal-transformation-zfs #:boot? #t))
   %os)
