* Guix System
#+name: guix-system
#+begin_src scheme :tangle live-desktop.scm :noweb yes
  (use-modules (nonguix transformations)
               (rosenthal)
               <<module>>)

  (define %os
    (operating-system
      (host-name "live-system")
      (bootloader <<bootloader>>)
      (file-systems %base-file-systems)
      (users (cons* <<user>> %base-user-accounts))
      (skeletons %rosenthal-skeletons)
      (packages
       (append (specs->pkgs+out <<package>>)
               (@@ (gnu system install) %installer-disk-utilities)
               %base-packages))
      (services
       (cons* <<service>>
              ((@@ (gnu system install) cow-store-service))
              %rosenthal-desktop-services))))

  ((compose <<transformation>>)
   %os)
#+end_src

      home-fontconfig-service-type
    (let ((sans  "Noto Sans")
          (serif "Noto Serif")
          (mono  "Victor Mono")
          (emoji "Noto Color Emoji"))
      `((alias
         (family "sans-serif")
         (prefer
          (family ,sans)
          (family "Noto Sans CJK CN")
          (family ,emoji)))
        (alias
         (family "serif")
         (prefer
          (family ,serif)
          (family "Noto Serif CJK CN")
          (family ,emoji)))
        (alias
         (family "monospace")
         (prefer
          (family ,mono)
          (family "Noto Serif CJK CN")
          (family ,emoji)))

        ,@(map (lambda (name)
                 `(alias
                   (family ,name)
                   (prefer
                    (family ,sans)
                    (family "sans-serif"))))
               '("BlinkMacSystemFont"
                 "-apple-system"
                 "system-ui"
                 "ui-sans-serif"))
        (alias
         (family "ui-serif")
         (prefer
          (family ,serif)
          (family "serif")))
        (alias
         (family "ui-monospace")
         (prefer
          (family ,mono)
          (family "monospace"))))))
#+end_src
