* Guix System
#+begin_src scheme :tangle worker.scm :noweb yes
  (load "../common.scm")
  (use-modules (rosenthal) <<module>>)

  (define (%os host-name system 32bit-support? workers threads-per-worker bios-boot)
    (operating-system
      (host-name host-name)
      (timezone "Etc/GMT-8")
      (bootloader
       (bootloader-configuration
         (bootloader
           (if bios-boot
               grub-bootloader
               grub-efi-removable-bootloader))
         (targets (list (or bios-boot "/efi")))
         (theme
          (grub-theme
            (inherit (grub-theme))
            (resolution #f)))))
      (kernel (spec->pkg "linux-libre@6.12"))
      (kernel-arguments
       (cons* "kernel.sysrq=0"
              "zswap.enabled=1"
              "zswap.max_pool_percent=90"
              %default-kernel-arguments))
      (initrd-modules
       (cons* "sd_mod"
              "virtio_scsi"
              %base-initrd-modules))
      (file-systems (append <<file-systems>> %base-file-systems))
      (swap-devices
       (list (swap-space
               (target (file-system-label "swap")))))
      (users (cons* <<user>> %base-user-accounts))
      (packages
       (append (specs->pkgs+out
                "mosh"
                "rsync"
                <<package>>)
               %base-packages))
      (services
       (cons* <<service>>
              (modify-services %base-services
                (guix-service-type
                 config => (guix-configuration
                             (inherit config)
                             (build-accounts (* 2 workers))
                             (substitute-urls
                              '("https://ci.guix.gnu.org"
                                "https://bordeaux.guix.gnu.org"))
                             (extra-options
                              (list "--cores" (number->string threads-per-worker))))))))))

  %os
#+end_src

** cuirass-remote-worker
Open TCP port 5558 and connect to server IPv4 address.
#+begin_src scheme :noweb-ref service
  (service cuirass-remote-worker-service-type
    (cuirass-remote-worker-configuration
      (publish-port 5558)
      (workers workers)
      (server "100.97.195.93:5555")       ;nuporta (Tailscale)
      (systems
       (cond
        ((string=? system "x86_64-linux")
         `("x86_64-linux"
           ,@(if 32bit-support? '("i686-linux") '())))
        ((string=? system "aarch64-linux")
         `("aarch64-linux"
           ,@(if 32bit-support? '("armhf-linux") '())))))
      (substitute-urls
       '("https://cache-cdn.guix.moe"
         "https://ci.guix.moe"
         "https://ci.guix.gnu.org"
         "https://bordeaux.guix.gnu.org"))))
#+end_src
