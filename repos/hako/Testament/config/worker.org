** Guix System
#+begin_src scheme :tangle ../tangled/worker/worker.scm :noweb yes
  (use-modules <<module>>)

  (lambda (host-name system 32bit-support? workers bios-boot)
    (operating-system
      (host-name host-name)
      (locale "en_US.UTF-8")
      (timezone "Etc/GMT-8")
      (bootloader
       (bootloader-configuration
         (bootloader
           (if bios-boot
               grub-bootloader
               grub-efi-removable-bootloader))
         (targets (list (or bios-boot "/efi")))))
      (kernel linux-server/dolly)
      (initrd-modules
       (cons* "sd_mod"
              (base-initrd-modules kernel)))
      (kernel-arguments
       (cons* <<kernel-argument>>
              %default-kernel-arguments))
      (file-systems
       (append <<file-systems>>
               %base-file-systems))
      (swap-devices
       (append <<swap-devices>>))
      (users
       (cons* <<user-account>>
              %base-user-accounts))
      (packages
       (append %testament-cli-packages
               %base-packages))
      (services
       (cons* <<service>>
              (modify-services %base-services
                ;; These services will be configured later.
                (delete guix-service-type))))
      (sudoers-file
       (plain-file "sudoers"
         (string-join
          (list (plain-file-content %sudoers-specification)
                <<sudoers>>)
          "\n")))))
#+end_src

** Continuous Integration
*** cuirass-remote-worker
Open TCP port 5558 and connect to server IPv4 address.
#+begin_src scheme :noweb-ref service
  (service cuirass-remote-worker-service-type
    (cuirass-remote-worker-configuration
      (publish-port 5558)
      (workers workers)
      (server "100.97.195.93:5555")       ;nuporta (Tailscale)
      (systems
       (cond
        ((string=? system "x86_64-linux")
         `("x86_64-linux"
           ,@(if 32bit-support? '("i686-linux") '())))
        ((string=? system "aarch64-linux")
         `("aarch64-linux"
           ,@(if 32bit-support? '("armhf-linux") '())))))
      (substitute-urls
       '("https://substitutes.nonguix.org"
         "https://ci.guix.gnu.org"
         "https://bordeaux.guix.gnu.org"))))
#+end_src
