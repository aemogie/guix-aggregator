** File Systems
Disk layout:
#+begin_example
  +---------+-----------------------+
  | ESP     | Btrfs subvolumes      |
  | /efi    |_ _ _ _ _ _ _ _ _ _ _ _|
  |         |                       |
  |         | LUKS encrypted volume |
  |         |_ _ _ _ _ _ _ _ _ _ _ _|
  |         |                       |
  |         | LVM Logical volume    |
  |_ _ _ _ _|_ _ _ _ _ _ _ _ _ _ _ _|
  |         |         |             |
  | part. 1 | part. 2 |             |
  |_ _ _ _ _| _ _ _ _ | disk 2      |
  |                   |             |
  | disk 1            |             |
  +-------------------+-------------+
#+end_example

Btrfs subvolumes:
#+begin_example
  @Library

Root file system layout:
#+begin_example
  /               tmpfs
  ├─/efi          fat32 ESP
  ├─/boot         btrfs System/@Boot
  ├─/etc          btrfs @Config
  ├─/gnu/store    btrfs System/@Store
  ├─/home         btrfs @Home
  ├─/swap         btrfs System/@Swap
  ├─/var/guix     btrfs System/@Guix
  ├─/var/lib      btrfs @Data
  └─/var/log      btrfs System/@Log
#+end_example

#+begin_src scheme :noweb yes :noweb-ref file-systems
  (let ((options-for-subvolume
         (lambda (subvolume)
           (string-join
            (list "compress=zstd:1"
                  (or (and=> subvolume (cut string-append "subvol=" <>))
                      "subvolid=5"))
            ","))))
    (map (match-lambda
           ((subvolume . mount-point)
            (file-system
              (device "/dev/mapper/Mentha-LUKS")
              (mount-point mount-point)
              (type "btrfs")
              (options (options-for-subvolume subvolume))
              (dependencies mapped-devices)
              (create-mount-point? #t)
              (check? (string=? mount-point "/gnu/store"))
              (needed-for-boot?
               (member mount-point '("/etc" "/gnu/store" "/var/guix"))))))
         '(("System/@Boot"  . "/boot")
           ("System/@Store" . "/gnu/store")
           ("System/@Swap"  . "/swap")
           ("System/@Guix"  . "/var/guix")
           ("System/@Log"   . "/var/log")
           ("@Config" . "/etc")
           ("@Home"   . "/home")
           ("@Data"   . "/var/lib")

** SOPS Secrets
Create GPG home directory specifically for use with SOPS.
#+begin_src shell
  MY_GPGHOME="<GPG homedir>"

  mkdir --parents --mode=0700 "$MY_GPGHOME"
  cat <<EOF > "$MY_GPGHOME/gpg.conf"
  pinentry-mode loopback
  EOF
  cat <<EOF > "$MY_GPGHOME/gpg-agent.conf"
  allow-loopback-pinentry
  EOF

  unset MY_GPGHOME
#+end_src

Generate passwordless ECC key:
#+begin_src shell
  MY_GPGHOME="<GPG homedir>"
  MY_NAME="<Your Name>"

  gpg --homedir "$MY_GPGHOME" --batch --generate-key <<EOF
  %no-protection
  Key-Type: EDDSA
  Key-Curve: ed25519
  Subkey-Type: ECDH
  Subkey-Curve: cv25519
  Name-Real: $MY_NAME
  Expire-Date: 0
  EOF

  unset MY_GPGHOME MY_NAME
#+end_src

Custom xref backend for Citre.  Arei's builtin xref backend provides =xref-buffer-location= but Citre expects =xref-file-location=.
#+begin_src scheme :noweb-ref emacs-package
  (emacs-package
    (name 'arei)
    (package (spec->pkg "emacs-arei"))
    (load-after-packages '(xref))
    (load-force? #t)
    (extra-after-load
     (list #%(defun hako/arei--xref-backend () 'hako/arei)
           #%(defun hako/arei-xref--make-xref-loc (identifier)
               (when-let* ((info (arei-xref--request-lookup identifier))
                           (file (arei-nrepl-dict-get info "file"))
                           (line (arei-nrepl-dict-get info "line"))
                           (column (arei-nrepl-dict-get info "column")))
                          (xref-make-file-location file line column)))
           #%(cl-defmethod xref-backend-identifier-at-point ((_backend (eql hako/arei)))
                           (thing-at-point 'symbol t))
           #%(cl-defmethod xref-backend-definitions ((_backend (eql hako/arei)) identifier)
                           (when (arei-connected-p)
                             (when-let* ((loc (hako/arei-xref--make-xref-loc identifier)))
                                        (list (xref-make identifier loc))))))))
#+end_src

In Emacs, run ~arei-connect~ (=C-c C-a=) or ~sesman-start~ (=C-c C-s C-s=) to connect to the server.
